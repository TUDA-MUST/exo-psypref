% Changes against fig2.tex:
%   - the arrow explained in the first panel: above its tail, "<marker> CMA-ES at 40" in the
%     comparator's text colour, so that the arrow reads from that level to the number at the dot;
%   - the comparator named in every panel: by that gloss (first panel), by "<triangle> Arens' GP"
%     under its curve where the comparator differs from the first panel's (3-35 Nm), and by the
%     direct label right of the last panel, now that panel's comparator only ("CMA-ES, 8 weights"),
%     not "CMA-ES or Arens' GP, 8 weights";
%   - panel titles name what the setting fixes ("3-12 Nm peaks", ..., "Fixed 12 Nm peaks"); the
%     second line says "Generic assistance 2.6 Nm away", not "... off";
%   - "~20 min" right of the k = 24 rule in the first panel (k_users x seconds per comparison);
%   - the clipped start of CMA-ES on Schaefer's family as "Start 23", not "(down arrow) 23 %".
% Layout (pt): three curve panels 3-12 / 3-35 / fixed 12 Nm, axes 85 x 70 at x = 35, 130, 225,
% y = 21, 40-100 %; direct labels in one column right of the third panel at the heights the script
% spread for that panel's line ends, each joined to its line's end by a thin leader; the fourth
% panel: axes 62 x 70 at x = 381, y = 21, 0-100 % against k = 1-40; the fifth: four stacked bars
% 0-100 % of users, x = 452-502, 7.5 pt high, 15 pt apart, the method's name above each, the better
% share in its coloured segment, the worse share in a column right of the bars, the key in the row of
% the x labels.

\def\ftpanel#1#2#3#4#5{%

    \node[ft lab, icMuted, anchor=base west] at ([yshift=2.0pt]axis cs:0.4,\icv{ftp}{#1}{ceiling})
      {Best 8-weight profile};
    \fi
    \draw[ic generic] (axis cs:0,50) -- (axis cs:40,50);
    % 95 % bands: PsyPref and the comparator
    \addplot[name path=plo, draw=none, forget plot] table[x=k, y=psypref_lo] {#2};
    \addplot[name path=phi, draw=none, forget plot] table[x=k, y=psypref_hi] {#2};

    \addplot[name path=bhi, draw=none, forget plot] table[x=k, y=best_hi] {#2};

    % the takeaway: the comparator's level after 40, reached by PsyPref at the first k

    % the comparator's markers once more, over the arrow

    % the arrow explained (first panel): its tail is the comparator after 40 comparisons; anchor

    \fi
    % the minutes of walking at k_users, right of the rule at the bottom (first panel)
    \ifnum\icv{ftp}{#1}{min20}=1
    \node[ft lab, icMuted, anchor=base west, inner sep=0pt]

    \ifnum\icv{ftp}{#1}{cmplab}=1

       \hphantom{\ftcmpmark{\icv{ftp}{#1}{cmp_code}}\hspace{1.8pt}}8 weights}
